import math
import numpy as np

def calc_spur(TOOTH: int, MODULE: int, PRESSURE_ANGLE: int = 20) -> dict:
    """
    spur gear 설계에 필요한 값을 계산합니다.
    PCD : Pitch Circle Diameter
    BCD : Base Circle Diameter
    ACD : Addendum Circle Diameter
    DCD : Dedendum Circle Diameter
    """
    
    Z_min = 2 / (np.sin(np.radians(PRESSURE_ANGLE)) ** 2)
    if TOOTH < Z_min:
        raise ValueError(
            f"설정한 잇수가 최소 잇수({math.ceil(Z_min)}개)보다 작습니다."
        )

    PCD = MODULE * TOOTH
    BCD = PCD * np.cos(np.radians(PRESSURE_ANGLE))
    CIRCULAR_PITCH = np.pi * PCD / TOOTH
    ACD = PCD + 2 * MODULE
    DCD = PCD - 2.5 * MODULE
    
    info = {
        "PCD": PCD,
        "BCD": BCD,
        "CIRCULAR_PITCH": CIRCULAR_PITCH,
        "ACD": ACD,
        "DCD": DCD,
    }
    
    return info

  
  

print(calc_spur(24, 2, 20))
