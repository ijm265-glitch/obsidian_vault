CASE nMainSeq OF
    0: (* A 모듈 동작 요청 *)
        bStart_A := TRUE; (* Req ON *)
        
        IF bDone_A THEN (* Ack ON 확인 *)
            bStart_A := FALSE; (* Req OFF *)
            nMainSeq := 10;
        END_IF;

    10: (* A 모듈의 Ack가 꺼진 것을 확인하고 B 모듈 동작 요청 *)
        IF NOT bDone_A THEN
            bStart_B := TRUE; (* Req ON *)
            IF bDone_B THEN
                bStart_B := FALSE;
                nMainSeq := 0;
            END_IF;
        END_IF;
END_CASE;
